            \begin{frame}{Inflación}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] La etapa de Inflación se define como un período de expansión acelerada y permite resolver los problemas del horizonte, planitud y sobrepoblación de defectos topológicos.
                    \item[\textcolor{brown}{$\bullet$}] El modelo inflacionario más sencillo considera la dinámica de un único campo: el inflatón.
                        \begin{equation*}
                            S_m = - \int \sqrt{-g} \left[ \frac{1}{2} g^{\mu\nu} \partial_\mu \phi \partial_\nu \phi + V(\phi) \right] ~ d^4 x
                        \end{equation*}
                        \begin{itemize}
                            \item[\textcolor{cyan}{$\bullet$}] De esta acción se deriva el tensor de energía-momento
                                \begin{equation*}
                                    T_{\mu\nu} = \partial_\mu \phi \partial_\nu \phi - g_{\mu\nu} \left[ \frac{1}{2} g^{\alpha\beta} \partial_\alpha \phi \partial_\beta \phi + V(\phi) \right]
                                \end{equation*}
                            \item[\textcolor{cyan}{$\bullet$}] Comparando con un fluido ideal se deriva
                                \begin{equation*}
                                    \rho = \frac{1}{2} \dot{\phi}^2 + V(\phi) \hspace{1.5cm} P = \frac{1}{2} \dot{\phi}^2 - V(\phi)
                                \end{equation*}
                            \begin{itemize}
                                \item[\textcolor{red}{$\Longrightarrow$}] La condición de \textit{slow-roll} es $\dot{\phi}^2 \ll V(\phi)$, cuando se cumple la presión efectiva se vuelve negativa $(P = - \rho)$, generando una expansión acelerada del universo.
                            \end{itemize}
                        \end{itemize}  
                \end{itemize}
            \end{frame}

        \subsection{Ondas Gravitacionales}

            \begin{frame}{Ondas Gravitacionales}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Las ondas gravitacionales (GWs) surgen como soluciones a las ecuaciones de Einstein linealizadas al introducir una perturbación tensorial en la métrica
                        \begin{equation*}
                            g_{\mu\nu} = \bar{g}_{\mu\nu} + h_{\mu\nu}
                        \end{equation*}
                        \vspace{-0.5cm}
                        \begin{itemize}
                            \item[\textcolor{cyan}{$\Longrightarrow$}] La métrica de fondo estará descripta por la métrica de FLRW plana $(ds^2 = - dt^2 + a^2 \, \delta_{ij} \, dx^i \, dx^j)$
                        \end{itemize}
                    \item[\textcolor{brown}{$\bullet$}] A través del \textit{gauge} TT (\textit{Transverse-Traceless}) la perturbación se escribe como suma de dos polarizaciones
                        \small{\begin{equation*}
                            h_{\mu\nu} = \begin{pmatrix}
                                0 & 0 & 0 & 0\\
                                0 & h_+ & h_\times & 0\\
                                0 & h_\times & - h_+ & 0\\
                                0 & 0 & 0 & 0 
                            \end{pmatrix} = \hspace{3.75cm}
                        \end{equation*}}
                \end{itemize}

                \begin{textblock*}{5cm}(8cm,5.5cm)
                    \includegraphics[scale = 0.175]{Figures/Polarizaciones_GWs.png}
                \end{textblock*}
            \end{frame}

            \begin{frame}{GWs en un fondo cosmológico}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Un fondo cosmológico utiliza una distribución de energía-materia para las ondas gravitacionales dada por
                    \begin{equation*}
                        T^{GW}_{\mu\nu} = \frac{1}{32 \pi G} \left\langle \nabla_\mu h_{\alpha \beta} \nabla_\nu h^{\alpha \beta} \right\rangle
                    \end{equation*}
                    \begin{itemize}
                        \item[\textcolor{cyan}{$\Longrightarrow$}] con lo que se puede definir a la densidad de energía fraccionaria para las ondas gravitacionales como
                        \begin{equation*}
                            \Omega_{\text{GW}} = \frac{\rho_{GW}}{\rho_c} = \frac{1}{32 \pi G \rho_c} \left\langle \dot{h}_{\alpha \beta} \dot{h}^{\alpha \beta} \right\rangle
                        \end{equation*}
                        \item[\textcolor{cyan}{$\Longrightarrow$}] Y al espectro de potencias para la densidad de energía fraccionaria a partir de
                        \begin{equation*}
                            \Omega_{GW} = \int \Omega_{GW} (k) \, d \log k
                        \end{equation*}
                    \end{itemize}
                \end{itemize}
            \end{frame}

            \begin{frame}{Ecuación de ondas en un fondo cosmológico}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Las ondas se pueden escribir como
                        \begin{equation*}
                            h_{ij}^{TT} (\textbf{k}, t) = \Lambda_{ij,kl} h_{ij} \left( \textbf{k}, t \right) = \sum_{r = +, \times} h_r (\textbf{k}, t) \, \hat{e}_{ij}^r \left( \hat{\textbf{k}} \right)
                        \end{equation*}
                    \item[\textcolor{brown}{$\bullet$}] La de Einstein para una métrica FLRW perturbada da la ecuación de ondas
                        \begin{equation*}
                            h_{r}^{\prime \prime} (\textbf{k}, \eta) + 2 \mathcal{H} \, h_{r}^\prime (\textbf{k}, \eta) + k^2 h_{r} (\textbf{k}, \eta) = \frac{1}{m_p^2} \Pi_{r}^{TT} (\textbf{k}, \eta)
                        \end{equation*}
                    \begin{itemize}
                        \item[\textcolor{cyan}{$\Longrightarrow$}] Donde $\Pi$ representa a las perturbaciones tensoriales en el tensor de energía-momento
                            \begin{equation*}
                                \Pi_{ij} = T_{ij} - a^2 P \left( \delta_{ij} + h_{ij} \right)
                            \end{equation*}
                    \end{itemize}
                \end{itemize}
            \end{frame}

            \begin{frame}{Fondos estocásticos}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Los Fondos Estocásticos de Ondas Gravitacionales (SGWB) se corresponden a superposición incoherente de ondas producidas por distintos procesos globales y no aislados
                    \item[\textcolor{brown}{$\bullet$}] La detección de estos fondos se basa en buscar correlaciones en las señales medidas por múltiples detectores o puntos del espacio-tiempo
                    \item[\textcolor{brown}{$\bullet$}] Toda la información física de un SGWB gaussiano queda completamente codificada en su función de correlación de dos puntos espectral
                        \begin{equation*}
                            \left\langle h_r (\textbf{k}, \eta) \, h_{r^\prime}^* (\textbf{k}^\prime, \eta) \right\rangle = \frac{8 \pi^5}{k^3} h_c^2 (k, \eta) \, \delta (\textbf{k}^\prime - \textbf{k}) \, \delta_{rr^\prime} 
                        \end{equation*}
                    \item[\textcolor{brown}{$\bullet$}] La densidad fraccionaria de ondas gravitacionales puede ser determinado a partir de
                        \begin{equation*}
                            \Omega_{GW} (f) = \frac{2 \pi^2 f^2 h_c^2 (f)}{3 H_0^2}
                        \end{equation*}
                \end{itemize}
            \end{frame}

            \begin{frame}{Detección de SGWB}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] La detección de estos fondos se basa en buscar correlaciones en las señales medidas por múltiples detectores o puntos del espacio-tiempo
                    \item[\textcolor{brown}{$\bullet$}] La señal medida por cada detector se puede descomponer como
                        \begin{equation*}
                            s(t) = h(t) + n(t)
                        \end{equation*}
                    \item[\textcolor{brown}{$\bullet$}] Asumiendo que las señales en varios detectores no está correlacionada entre sí ni con el ruido, se puede obtener la señal del espectro del fondo de ondas gravitacionales:
                        \begin{equation*}
                            \begin{split}
                                \left\langle s_1 (t) \, s_2 (t) \right\rangle &= \left\langle \left[ h_1 (t) + n_1 (t) \right] \left[ h_2 (t) + n_2 (t) \right] \right\rangle\\
                                &= \left\langle h_1 (t) \, h_2 (t) \right\rangle + \left\langle n_1 (t) \, n_2 (t) \right\rangle\\
                                &= \left\langle h_1 (t) \, h_2 (t) \right\rangle
                            \end{split}
                        \end{equation*}
                \end{itemize}
            \end{frame}
