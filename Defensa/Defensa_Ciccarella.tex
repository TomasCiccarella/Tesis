            \begin{frame}{Inflación}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] La etapa de Inflación se define como un período de expansión acelerada y permite resolver los problemas del horizonte, planitud y sobrepoblación de defectos topológicos.
                    \item[\textcolor{brown}{$\bullet$}] El modelo inflacionario más sencillo considera la dinámica de un único campo: el inflatón.
                        \begin{equation*}
                            S = - \int \sqrt{-g} \left[ \frac{1}{2} g^{\mu\nu} \partial_\mu \phi \partial_\nu \phi + V(\phi) \right] ~ d^4 x
                        \end{equation*}
                        \begin{itemize}
                            \item[\textcolor{cyan}{$\bullet$}] De esta acción se deriva el tensor de energía-momento
                                \begin{equation*}
                                    T_{\mu\nu} = \partial_\mu \phi \partial_\nu \phi - g_{\mu\nu} \left[ \frac{1}{2} g^{\alpha\beta} \partial_\alpha \phi \partial_\beta \phi + V(\phi) \right]
                                \end{equation*}
                            \item[\textcolor{cyan}{$\bullet$}] Comparando con un fluido ideal se deriva
                                \begin{equation*}
                                    \rho = \frac{1}{2} \dot{\phi}^2 + V(\phi) \hspace{1.5cm} P = \frac{1}{2} \dot{\phi}^2 - V(\phi)
                                \end{equation*}
                            \begin{itemize}
                                \item[\textcolor{red}{$\Longrightarrow$}] Si $\dot{\phi}^2 \ll V(\phi)$ la presión efectiva se vuelve negativa, generando una expansión acelerada del universo. A esta condición se la conoce como \textit{slow-roll}.
                            \end{itemize}
                        \end{itemize}  
                \end{itemize}
            \end{frame}

            \begin{frame}{Inflación: \textit{slow-roll}}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Durante la Inflación consideraremos que el inflatón ``rueda'' lentamente hasta que $\dot{\phi}^2 \simeq V(\phi)$. 
                    \item[\textcolor{brown}{$\bullet$}] Se definen
                        \vspace{-0.1cm}
                        \begin{equation*}
                            \epsilon = \frac{m_p^2}{2} \left[ \frac{V^\prime (\phi)}{V(\phi)} \right]^2 \hspace{1.5cm} \eta = m_p^2 \frac{V^{\prime \prime} (\phi)}{V(\phi)}
                        \end{equation*}
                        \begin{itemize}
                            \item[\textcolor{cyan}{$\bullet$}] Bajo la aproximación de \textit{slow-roll} $\epsilon, \eta \ll 1$

                    \item[\textcolor{brown}{$\bullet$}] Las ondas gravitacionales (GWs) surgen como soluciones a las ecuaciones de Einstein linealizadas al introducir una perturbación tensorial en la métrica

                            g_{\mu\nu} = \bar{g}_{\mu\nu} + h_{\mu\nu}

                            h_{\mu\nu} = \begin{pmatrix}
                                0 & 0 & 0 & 0\\
                                0 & h_+ & h_\times & 0\\
                                0 & h_\times & - h_+ & 0\\
                                0 & 0 & 0 & 0 
                            \end{pmatrix} = \hspace{3.75cm}
                        \end{equation*}}
                \end{itemize}

            \begin{frame}{GWs en un fondo cosmológico}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Un fondo cosmológico utiliza una distribución de energía-materia para las ondas gravitacionales dada por
                    \begin{equation*}
                        T^{GW}_{\mu\nu} = \frac{1}{32 \pi G} \left\langle \nabla_\mu h_{\alpha \beta} \nabla_\nu h^{\alpha \beta} \right\rangle \hspace{0.2cm} \textcolor{cyan}{\Longrightarrow} \hspace{0.2cm} \rho_{GW} = \frac{1}{32 \pi G} \left\langle \dot{h}_{\alpha \beta} \dot{h}^{\alpha \beta} \right\rangle
                    \end{equation*} 
                    \item[\textcolor{brown}{$\bullet$}] La ecuación de ondas para un fondo tipo FLRW es
                        \begin{equation*}
                            \left( h_{ij}^{TT} \right)^{\prime \prime} (\textbf{k}, \eta) + 2 \mathcal{H} \left( h_{ij}^{TT} \right)^\prime (\textbf{k}, \eta) + k^2 h_{ij}^{TT} (\textbf{k}, \eta) = \frac{2}{m_p^2} \Pi_{ij}^{TT} (\textbf{k}, \eta)
                        \end{equation*}
                        \begin{itemize}
                            \item[\textcolor{cyan}{$\bullet$}] El término de Hubble puede ser absorbido en un reescalado del campo de la forma $H_{ij} = a h_{ij}^{TT}$, y la ecuación para la dinámica de cada polarización $r = +, \times$ es 
                                \begin{equation*}
                                    H^{\prime \prime}_r (\textbf{k}, \eta) + \left( k^2 - \frac{a^{\prime \prime}}{a} \right) H_r (\textbf{k}, \eta) = \frac{a}{m_p^2} \Pi_r (\textbf{k}, \eta)
                                \end{equation*} 
                        \end{itemize}
                \end{itemize}
            \end{frame}

            \begin{frame}{Fondos estocásticos}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Los Fondos Estocásticos de Ondas Gravitacionales (SGWB) se corresponden a superposición incoherente de ondas producidas por distintos procesos globales y no aislados
                    \item[\textcolor{brown}{$\bullet$}] La detección de estos fondos se basa en buscar correlaciones en las señales medidas por múltiples detectores o puntos del espacio-tiempo
                    \item[\textcolor{brown}{$\bullet$}] Toda la información física de un SGWB gaussiano queda completamente codificada en su función de correlación de dos puntos espectral
                        \begin{equation*}
                            \left\langle h_r (\textbf{k}, \eta) \, h_{r^\prime}^* (\textbf{k}^\prime, \eta) \right\rangle = \frac{8 \pi^5}{k^3} h_c^2 (k, \eta) \, \delta (\textbf{k}^\prime - \textbf{k}) \, \delta_{rr^\prime} 
                        \end{equation*}
                    \item[\textcolor{brown}{$\bullet$}] La densidad fraccionaria de ondas gravitacionales puede ser determinado a partir de
                        \begin{equation*}
                            h_c^2 (f) = \frac{3 H_0^2}{2 \pi^2 f^2} \Omega_{GW} (f)
                        \end{equation*}
                \end{itemize}
            \end{frame}

            \begin{frame}{Detección de SGWB}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] La detección de estos fondos se basa en buscar correlaciones en las señales medidas por múltiples detectores o puntos del espacio-tiempo
                    \item[\textcolor{brown}{$\bullet$}] La señal medida por cada detector se puede descomponer como
                        \begin{equation*}
                            s(t) = h(t) + n(t)
                        \end{equation*}
                    \item[\textcolor{brown}{$\bullet$}] Asumiendo que las señales en varios detectores no está correlacionada entre sí ni con el ruido, se puede obtener la señal del espectro del fondo de ondas gravitacionales:
                        \begin{equation*}
                            \begin{split}
                                \left\langle s_1 (t) \, s_2 (t) \right\rangle &= \left\langle \left[ h_1 (t) + n_1 (t) \right] \left[ h_2 (t) + n_2 (t) \right] \right\rangle\\
                                &= \left\langle h_1 (t) \, h_2 (t) \right\rangle + \left\langle n_1 (t) \, n_2 (t) \right\rangle\\
                                &= \left\langle h_1 (t) \, h_2 (t) \right\rangle
                            \end{split}
                        \end{equation*}
                \end{itemize}
            \end{frame}

        \subsection{Resonancia Paramétrica}

            \begin{frame}{Resonancia Paramétrica}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Durante \textit{Preheating} el inflatón oscila alrededor del mínimo del potencial, transfiriendo energía a otros campos $\chi$
                    \begin{itemize}
                        \item[\textcolor{cyan}{$\Longrightarrow$}] A este fenómeno se lo conoce como resonancia paramétrica 
                    \end{itemize}
                    \item[\textcolor{brown}{$\bullet$}] La dinámica está regida por
                        \small{\begin{equation*}
                            S = \int \sqrt{-g} \left[ \frac{m_p^2}{2} R - \frac{1}{2} \partial_\mu \phi \partial^\mu \phi - \frac{1}{2} \partial_\mu \chi \partial^\mu \chi - V(\phi) - \frac{1}{2} g^2 \phi^2 \chi^2 \right] \, d^4 x 
                        \end{equation*}}
                    \begin{itemize}
                        \item[\textcolor{cyan}{$\Longrightarrow$}] Utilizando una métrica de tipo FLRW, las ecuaciones de los campos se escriben como
                            \begin{equation*}
                                \ddot{\phi} + 3 H \dot{\phi} - \frac{1}{a^2} \nabla^2 \phi + \frac{\partial V(\phi)}{\partial \phi} + g^2 \phi \chi^2 = 0
                            \end{equation*}
                            \begin{equation*}
                                \ddot{\chi} + 3 H \dot{\chi} - \frac{1}{a^2} \nabla^2 \chi + g^2 \phi^2 \chi = 0
                            \end{equation*}
                    \end{itemize}
                    \item[\textcolor{brown}{$\bullet$}] Para trabajar estas ecuaciones se hará un desarrollo perturbativo de la forma: $\phi = \bar{\phi} + \delta \phi$ y $\chi = \delta \chi$
                \end{itemize}
            \end{frame}

        \subsection{¿Qué es?}

            \begin{frame}{¿Qué es?}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] \CosmoLattice es un código público escrito en \texttt{C++} 
                    \item[\textcolor{brown}{$\bullet$}] Permite simular la dinámica no lineal de campos clásicos en un universo en expansión
                    \item[\textcolor{brown}{$\bullet$}] La dinámica general está descripta por
                        \footnotesize{\begin{equation*}
                            \begin{split}
                                S = - \int \sqrt{-g} & \left[ \sum_n \frac{1}{2} \partial_\mu \phi_n \partial^\mu \phi_n + \sum_m \left( D_\mu \varphi_m \right)^* \left( D^\mu \varphi_m \right) + \sum_k \left( D_\mu \Phi_k \right)^\dagger \left( D^\mu \Phi_k \right) \right. \\
                                &+ \left. \frac{1}{4} F_{\mu\nu} F^{\mu\nu} + \frac{1}{2} \text{Tr} \left( G_{\mu\nu} G^{\mu\nu} \right) + V \left( \phi_n, \varphi_m, \Phi_k \right) \right] d^4 x 
                            \end{split}
                        \end{equation*}} 
                    \item[\textcolor{brown}{$\bullet$}] Se trabaja bajo una métrica de fondo de tipo FLRW plana
                        \small{\begin{equation*}
                            ds^2 = - a^{2 \alpha} (\tau) \, d \tau^2 + a^2 (\tau) \, \delta_{ij} \, d x^i \, d x^j
                        \end{equation*}}
                        \vspace{-0.75cm}
                    \begin{itemize}
                        \item[\textcolor{cyan}{$\Longrightarrow$}] $\alpha = 0$ representa a usar tiempo coordenado y $\alpha = 1$ corresponde a usar tiempo conforme
                        \begin{equation*}
                            \alpha = 3 \frac{p - 2}{p + 2} \hspace{2cm} \left( V(\phi) = \phi^{p} \right)
                        \end{equation*}
                    \end{itemize}
                \end{itemize}

            \end{frame}

            \begin{frame}{Ecuaciones dinámicas}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] La evolución de los campos está dada por
                        \begin{equation*}
                            \phi^{\prime \prime}_n + (3 - \alpha) \frac{a^{\prime}}{a} \phi^\prime_n - a^{2 \alpha - 2} \nabla^2 \phi_n + a^{2 \alpha} \frac{\partial V}{\partial \phi_n} = 0
                        \end{equation*}
                    \item[\textcolor{brown}{$\bullet$}] La evolución del universo está dada por
                        \begin{equation*}
                            \frac{a^{\prime \prime}}{a} - \alpha \left( \frac{a^\prime}{a} \right)^2 = \frac{a^{2 \alpha}}{6 m_p^2} (\rho - 3 P)
                        \end{equation*} 
                    \item[\textcolor{brown}{$\bullet$}] La presión y densidad de energía se calculan a partir de 
                        \begin{equation*}
                            \rho = \sum_n \left( K_n + G_n \right) + V \hspace{2cm} P = \sum_n \left( K_n - \frac{1}{3} G_n \right) - V
                        \end{equation*}
                        \begin{equation*}
                            K_n = \frac{1}{2 a^{2\alpha}} \left( \phi^\prime_n \right)^2 \hspace{2cm} G_n = \frac{1}{2a^2} \sum_{i = 1}^3 \left( \partial_i \phi_n \right)^2
                        \end{equation*}
                \end{itemize}
            \end{frame}

            \begin{frame}{Ecuaciones para GW}
                \begin{itemize}
                    \item[\textcolor{brown}{$\bullet$}] La evolución de las ondas se hace a través de un campo auxiliar $u_{ij}$
                        \begin{equation*}
                            h_{ij} \left( \textbf{k}, t \right) = \Lambda_{ij,kl} \left( \hat{\textbf{k}} \right) \, u_{kl} \left( \textbf{k}, t \right)
                        \end{equation*}
                    \item[\textcolor{brown}{$\bullet$}] La evolución del campo auxiliar está determinada por
                        \begin{equation*}
                            u_{ij}^{\prime \prime} + (3 - \alpha) \frac{a^\prime}{a} u_{ij}^\prime - a^{2\alpha - 2} \nabla^2 u_{ij} = \frac{2}{m_p^2} a^{2 \alpha} \Pi_{ij}^{eff}
                        \end{equation*}
                    \item<3->[\textcolor{brown}{$\bullet$}] El espectro de potencias para la densidad de energía de las ondas se calcula a partir de

                            \Omega_{GW} (n, t) = \frac{1}{\rho_c} \frac{m_p^2 l k_{IR}}{8 \pi} \frac{\delta x}{N^5} \#_l \left\langle \dot{h}_{ij} (n, t) \, \dot{h}^*_{ij} (n,t) \right\rangle  
                        \end{equation*}
                \end{itemize}
